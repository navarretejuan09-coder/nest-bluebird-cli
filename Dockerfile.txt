# Bluebird - NestJS Static Analysis CLI
# https://github.com/endpointclosing/bluebird

FROM node:22-alpine AS base

WORKDIR /app

# Copy package files and tsconfig for build
COPY package.json pnpm-lock.yaml pnpm-workspace.yaml tsconfig.json ./

# Activate the exact pnpm version pinned in package.json "packageManager"
RUN corepack enable && corepack install
COPY packages/bluebird/package.json packages/bluebird/tsconfig.json ./packages/bluebird/

# Install dependencies
RUN pnpm install --frozen-lockfile --prod=false

# Copy source

COPY packages/bluebird/ ./packages/bluebird/

# Build
RUN pnpm --filter bluebird run build

# Prune to production dependencies only (deterministic via lockfile)
RUN CI=true pnpm install --frozen-lockfile --prod --filter bluebird --ignore-scripts

# Production image — distroless contains only the Node.js runtime, no shell/package manager/OS packages
FROM gcr.io/distroless/nodejs22-debian12 AS production

WORKDIR /app

# Copy built output and pruned production node_modules from build stage
COPY --from=base /app/packages/bluebird/dist ./packages/bluebird/dist
COPY --from=base /app/packages/bluebird/package.json ./packages/bluebird/
COPY --from=base /app/node_modules ./node_modules
COPY --from=base /app/packages/bluebird/node_modules ./packages/bluebird/node_modules

USER nonroot

# Set working directory for analysis
WORKDIR /workspace

ENTRYPOINT ["/nodejs/bin/node", "/app/packages/bluebird/dist/cli.js"]
CMD ["--help"]

# Usage:
# docker build -t bluebird .
# docker run --rm -v $(pwd):/workspace bluebird
# docker run --rm -v $(pwd):/workspace bluebird --format json
# docker run --rm -v $(pwd):/workspace bluebird --diff main
